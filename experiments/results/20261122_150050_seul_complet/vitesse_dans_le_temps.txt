"""Speed paradox over time: do fast lineages speed up gradually while their population falls?"""
import csv, glob
import numpy as np
R = "/Users/jul/Documents/Dev/EvolutionGame/experiments/results"
rows_all = []
for d in sorted(glob.glob(f"{R}/20261122_15*_seul_*")):
    sp = d.split("_seul_")[1]
    for f in sorted(glob.glob(f"{d}/seed_*/timeline.csv"), key=lambda f: int(f.split("seed_")[1].split("/")[0])):
        s = int(f.split("seed_")[1].split("/")[0])
        r = list(csv.DictReader(open(f)))
        if float(r[-1]["population"]) == 0: continue
        t = np.array([float(x["tick"]) for x in r]); v = np.array([float(x["mean_speed"]) for x in r])
        p = np.array([float(x["population"]) for x in r]); fr = np.array([float(x["food_rate"]) for x in r])
        def win(a, b): m = (t >= a) & (t < b); return v[m].mean(), p[m].mean()
        late = t >= 20000
        rho = np.corrcoef(v[late], p[late])[0, 1]
        rows_all.append((sp, s, *win(10000, 20000), *win(40000, 60000), *win(100000, 120001), rho,
                         t[np.argmax(v > 2.5)] if (v > 2.5).any() else None))
print(f"{len(rows_all)} populations ; relevés tous les {int(t[1]-t[0])} ticks")
print("espèce     graine  vitesse 10-20k / 40-60k / 100-120k   population 10-20k / 40-60k / 100-120k   corr(v,pop) après 20k   v>2,5 dès")
for sp, s, v1, p1, v2, p2, v3, p3, rho, first in sorted(rows_all, key=lambda x: -x[6]):
    print(f"{sp:16s} {s:2d}   {v1:4.2f} / {v2:4.2f} / {v3:4.2f}              {p1:4.0f} / {p2:4.0f} / {p3:4.0f}                {rho:+.2f}        {'' if first is None else int(first)}")
fast = [x for x in rows_all if x[6] > 2.5]; slow = [x for x in rows_all if x[6] <= 2.5]
for name, g in (("rapides", fast), ("autres", slow)):
    print(f"{name}: n={len(g)}  vitesse {np.mean([x[2] for x in g]):.2f} -> {np.mean([x[4] for x in g]):.2f} -> {np.mean([x[6] for x in g]):.2f}"
          f"  population {np.mean([x[3] for x in g]):.0f} -> {np.mean([x[5] for x in g]):.0f} -> {np.mean([x[7] for x in g]):.0f}"
          f"  corr moyenne {np.mean([x[8] for x in g]):+.2f}")
allr = np.array([x[8] for x in rows_all])
print(f"corr(vitesse, population) dans le temps : moyenne {allr.mean():+.2f}, négative dans {(allr < 0).sum()}/{len(allr)}")
v3 = np.array([x[6] for x in rows_all]); p3 = np.array([x[7] for x in rows_all])
print(f"entre populations (100-120k) : corr(vitesse, population) = {np.corrcoef(v3, p3)[0,1]:+.2f}")
