--- lignee.py
"""What makes a lineage strong? Traits of each population evolved alone vs its results in the crossed duels."""
import csv, glob, json, sys
sys.path.insert(0, "/private/tmp/claude-501/-Users-jul-Documents-Dev-EvolutionGame/cd67667f-3761-42ec-aa1a-b5872c8c57be/scratchpad")
from analyse import winners, run, R
import numpy as np
TRAITS = ["population", "mean_speed", "mean_energy", "mean_size", "food_rate", "births", "mean_age_at_death",
          "blind_frac", "max_lineage_gen", "deaths_hunger"]
ALONE = {"complet": "150050", "faim": "150705", "bruit": "151340", "memoire": "151957"}

def traits(sp, seed):
    name = "complet" if sp == "complet" else f"complet_{sp}"
    f = f"{R}/20261122_{ALONE[sp]}_seul_{name}/seed_{seed}/timeline.csv"
    rows = list(csv.DictReader(open(f)))[-10:]
    if float(rows[-1]["population"]) == 0: return None
    return {t: np.mean([float(r[t]) for r in rows]) for t in TRAITS}

def rank(x):
    return np.argsort(np.argsort(x)).astype(float)

duels = []   # (sp, i, j, complet_won)
for sp in ("faim", "bruit", "memoire"):
    for k in range(4):
        d = run(f"duel_evolues_complet_{sp}") if k == 0 else run(f"croise{k}_complet_{sp}")
        seeds = json.load(open(f"{d}/run.json"))["seeds"]
        pair = {s: seeds[(i + k) % len(seeds)] for i, s in enumerate(seeds)}
        for s, w in winners(d).items():
            if w in "CX": duels.append((sp, s, pair[s], w == "C"))
print(f"{len(duels)} duels tranchés")
T = {}
for sp in ALONE:
    for s in range(1, 17):
        t = traits(sp, s)
        if t: T[(sp, s)] = t

print("\n1) Par duel : le camp au trait le plus élevé (mesuré seul) gagne-t-il ? (concordance, 50 % = aucun lien)")
for t in TRAITS:
    ok = n = 0
    for sp, i, j, cw in duels:
        a, b = T[("complet", i)][t], T[(sp, j)][t]
        if a == b: continue
        n += 1; ok += (a > b) == cw
    from math import comb
    k = max(ok, n - ok); p = min(1, 2 * sum(comb(n, x) for x in range(k, n + 1)) / 2 ** n)
    print(f"   {t:18s} {ok:3d}/{n} = {ok/n:4.0%}   p={p:.3f}")

print("\n2) Par lignée : score (part de duels gagnés) contre trait, Spearman")
score = {}
for sp, i, j, cw in duels:
    score.setdefault(("complet", i), []).append(cw)
    score.setdefault((sp, j), []).append(not cw)
keys = [k for k in score if len(score[k]) >= 2]
sc = np.array([np.mean(score[k]) for k in keys])
for t in TRAITS:
    x = np.array([T[k][t] for k in keys])
    rho = np.corrcoef(rank(x), rank(sc))[0, 1]
    # within-species: rank inside each species, then pooled
    print(f"   {t:18s} rho={rho:+.2f}  (n={len(keys)})")
print("\n   lignées (espèce, graine, score, vitesse, taille, population) triées par score :")
for k in sorted(keys, key=lambda k: np.mean(score[k])):
    print(f"   {k[0]:8s} {k[1]:2d}  {np.mean(score[k]):.2f} ({len(score[k]):2d})  v={T[k]['mean_speed']:.2f}  "
          f"taille={T[k]['mean_size']:.2f}  pop={T[k]['population']:.0f}  énergie={T[k]['mean_energy']:.0f}")

print("\n3) Concordance de la population (seule) par duel, espèce par espèce")
for sp in ("faim", "bruit", "memoire"):
    ok = n = 0; lost_bigger = 0
    for s, i, j, cw in duels:
        if s != sp: continue
        a, b = T[("complet", i)]["population"], T[(sp, j)]["population"]
        n += 1; ok += (a > b) == cw
        lost_bigger += (b > a) and cw
    print(f"   {sp:8s} {ok}/{n} = {ok/n:.0%}   ({sp} plus peuplé mais battu : {lost_bigger})")
fast = [k for k in keys if T[k]["mean_speed"] > 2.5]
print(f"\n4) Lignées rapides (vitesse > 2,5) : {len(fast)}, score moyen {np.mean([np.mean(score[k]) for k in fast]):.2f}, "
      f"population seule {np.mean([T[k]['population'] for k in fast]):.0f} contre "
      f"{np.mean([T[k]['population'] for k in keys if k not in fast]):.0f} pour les autres")
