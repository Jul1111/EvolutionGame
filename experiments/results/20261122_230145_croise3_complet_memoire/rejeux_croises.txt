--- analyse.py
"""Winner per seed of duel runs: replays (same pair, other chance) and crossed duels (complet i vs other j)."""
import csv, glob, json, os, sys
R = "/Users/jul/Documents/Dev/EvolutionGame/experiments/results"

def winners(d):
    out = {}
    for f in glob.glob(f"{d}/seed_*/timeline.csv"):
        s = int(f.split("seed_")[1].split("/")[0])
        rows = list(csv.DictReader(open(f)))
        last = rows[-1]
        pops = {k[5:]: float(v) for k, v in last.items() if k.startswith("pop__")}
        alive = [k for k, v in pops.items() if v > 0]
        out[s] = "E" if not alive else ("C" if alive == ["complet"] else "X" if len(alive) == 1 else "=")
    return out

def run(pattern):
    ds = sorted(glob.glob(f"{R}/*_{pattern}"))
    return ds[-1] if ds else None

for sp in ("faim", "bruit", "memoire"):
    print(f"\n== complet vs {sp}")
    reps = [run(f"duel_evolues_complet_{sp}")] + [run(f"rejeu{k}_complet_{sp}") for k in (1, 2)]
    W = [winners(d) if d else {} for d in reps]
    same = tot = 0
    for s in range(1, 17):
        line = [w.get(s, "-") for w in W]
        dec = [x for x in line if x in "CX"]
        if len(dec) >= 2:
            tot += 1; same += len(set(dec)) == 1
        print(f"  graine {s:2d}  orig/rejeu1/rejeu2 : {' '.join(line)}")
    print(f"  même gagnant à chaque rejeu : {same}/{tot}")
    # Crossed: matrix complet seed i vs other seed pair
    cross = {}
    for k in range(0, 4):
        d = reps[0] if k == 0 else run(f"croise{k}_complet_{sp}")
        if not d: continue
        seeds = json.load(open(f"{d}/run.json"))["seeds"]
        pair = {s: seeds[(i + k) % len(seeds)] for i, s in enumerate(seeds)}
        for s, w in winners(d).items():
            cross[(s, pair[s], k)] = w
    if cross:
        print("  croisés : par population complet (i) et par adversaire (j), victoires C/X")
        for role, idx in (("complet", 0), (sp, 1)):
            stats = {}
            for key, w in cross.items():
                if w in "CX":
                    st = stats.setdefault(key[idx], [0, 0]); st[0 if w == "C" else 1] += 1
            print(f"   {role:8s}", " ".join(f"{p}:{a}/{a+b}" for p, (a, b) in sorted(stats.items())))

import sys, math
sys.path.insert(0, "/private/tmp/claude-501/-Users-jul-Documents-Dev-EvolutionGame/cd67667f-3761-42ec-aa1a-b5872c8c57be/scratchpad")
from analyse import winners, run
import json
def binom_p(k, n):
    from math import comb
    lo = min(k, n - k)
    p = sum(comb(n, i) for i in range(0, lo + 1)) / 2 ** n * 2
    return min(1, p)
allC = allN = 0
for sp in ("faim", "bruit", "memoire"):
    cross = []
    for k in range(4):
        d = run(f"duel_evolues_complet_{sp}") if k == 0 else run(f"croise{k}_complet_{sp}")
        seeds = json.load(open(f"{d}/run.json"))["seeds"]
        pair = {s: seeds[(i + k) % len(seeds)] for i, s in enumerate(seeds)}
        for s, w in winners(d).items():
            if w in "CX": cross.append((s, pair[s], k, w == "C"))
    n = len(cross); c = sum(x[3] for x in cross)
    nk = [sum(1 for x in cross if x[2] == k) for k in range(4)]
    ck = [sum(x[3] for x in cross if x[2] == k) for k in range(4)]
    print(f"{sp}: complet gagne {c}/{n} p={binom_p(c,n):.3f}  par décalage {[f'{a}/{b}' for a,b in zip(ck,nk)]}")
    allC += c; allN += n
    # leave-one-out predictors
    def loo(idx):
        ok = tot = 0
        for j, x in enumerate(cross):
            others = [y[3] for i, y in enumerate(cross) if i != j and y[idx] == x[idx]]
            if not others: continue
            m = sum(others) / len(others)
            if m == 0.5: continue
            tot += 1; ok += (m > 0.5) == x[3]
        return ok, tot
    base = max(c, n - c) / n
    for name, idx in (("lignée complet", 0), ("lignée adverse", 1)):
        ok, tot = loo(idx); print(f"   prédit par {name} (sans le duel) : {ok}/{tot} = {ok/tot:.0%}  (toujours la majorité : {base:.0%})")
    # both: sum of scores
    ok = tot = 0
    for j, x in enumerate(cross):
        a = [y[3] for i, y in enumerate(cross) if i != j and y[0] == x[0]]
        b = [y[3] for i, y in enumerate(cross) if i != j and y[1] == x[1]]
        if not a or not b: continue
        sc = sum(a)/len(a) + sum(b)/len(b) - 1
        if sc == 0: continue
        tot += 1; ok += (sc > 0) == x[3]
    print(f"   prédit par les deux : {ok}/{tot} = {ok/tot:.0%}")
print(f"ensemble: {allC}/{allN} p={binom_p(allC, allN):.4f}")
